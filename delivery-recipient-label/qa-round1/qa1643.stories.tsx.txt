import type { Meta, StoryObj } from "@storybook/react";
import type { FC } from "react";
import type {
	TeamBridgeDeliveryState,
	TeamBridgeMessage,
} from "../../../../shared/team-bridge-contract";
import { TeamBridgeMessageList } from "./team-bridge-message-list";

/*
 * QA-ONLY (PR #1643 round 1, never committed). The same fixtures are dropped
 * into the head worktree and a base (2ff3187) worktree so the frames compare
 * like for like: each story draws one recipient-label case as refused, held
 * (sender view) and screened-pending lines under one message.
 */
const lines = (
	endpoint: string,
	recipient_label: string,
	recipient_owner_label: string,
): TeamBridgeDeliveryState[] => [
	{ endpoint_id: `${endpoint}_r`, recipient_label, recipient_owner_label, state: "refused" },
	{ endpoint_id: `${endpoint}_h`, recipient_label, recipient_owner_label, state: "held" },
	{
		endpoint_id: `${endpoint}_p`,
		recipient_label,
		recipient_owner_label,
		state: "held",
		hold_class: "delivery_pending",
	},
];

const message = (delivery_states: TeamBridgeDeliveryState[]): TeamBridgeMessage => ({
	id: "bcmsg_qa1643",
	conversation_id: "bcv_qa1643",
	seq: 1,
	sender: { account_id: "acct_me", kind: "human", display_label: "Ada" },
	body: "QA 1643: can you pull the Q3 renewal list?",
	origin: "human",
	created_at: "2026-11-03T12:00:00.000Z",
	delivery_states,
});

const Thread: FC<{ states: TeamBridgeDeliveryState[] }> = ({ states }) => (
	<div data-qa="thread" className="flex w-[760px] flex-col bg-surface pt-6 text-ink">
		<TeamBridgeMessageList
			messages={[message(states)]}
			labelFor={(sender) => sender.display_label}
			tenantId="tnt_story"
		/>
	</div>
);

const meta = {
	title: "QA 1643/Delivery recipient",
	component: Thread,
	parameters: { layout: "centered" },
} satisfies Meta<typeof Thread>;
export default meta;
type Story = StoryObj<typeof meta>;
