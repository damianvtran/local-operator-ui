/** Everything a user could perceive about the band at one instant. */
function uxRead(cdp) {
	return cdp.evaluate(`(() => {
		const flat = (el) => (el ? el.textContent.replace(/\\s+/g, " ").trim() : null);
		const field = document.querySelector('textarea[aria-label="Message"]');
		const panel = document.querySelector("[data-lo-aside-panel]");
		const send = document.querySelector('button[aria-label="Send message"]');
		const a = document.activeElement;
		const region = panel
			? Array.from(panel.querySelectorAll("div")).find((n) => n.style && n.style.maxHeight !== "")
			: null;
		const lastTurn = region ? region.lastElementChild : null;
		const clip = region ? region.getBoundingClientRect() : null;
		const lt = lastTurn ? lastTurn.getBoundingClientRect() : null;
		const adopt = panel
			? Array.from(panel.querySelectorAll("button")).find((b) => b.textContent.trim().startsWith("Add to conversation"))
			: null;
		return {
			placeholder: field ? field.placeholder : null,
			fieldValue: field ? field.value : null,
			readOnly: field ? field.readOnly : null,
			active: a
				? a.tagName.toLowerCase() +
					(a.getAttribute("aria-label") ? "[" + a.getAttribute("aria-label") + "]" : "") +
					(a.tagName === "BUTTON" ? "{" + a.textContent.replace(/\\s+/g, " ").trim().slice(0, 30) + "}" : "")
				: null,
			activeIsField: a === field,
			activeIsBody: a === document.body,
			panel: panel !== null,
			panelText: flat(panel),
			panelAnnounce: panel ? flat(panel.querySelector("output")) : null,
			panelAlert: panel ? flat(panel.querySelector('[role="alert"]')) : null,
			adopt: adopt
				? {
						disabled: adopt.disabled,
						describedBy: adopt.getAttribute("aria-describedby"),
						title: adopt.getAttribute("title"),
					}
				: null,
			send: send ? { disabled: send.disabled, label: send.getAttribute("aria-label") } : null,
			region: region
				? {
						scrollTop: Math.round(region.scrollTop),
						scrollHeight: region.scrollHeight,
						clientHeight: region.clientHeight,
						turns: region.children.length,
						lastTurnTopVisible: lt ? lt.top >= clip.top - 1 && lt.top <= clip.bottom : null,
						lastTurnBottomVisible: lt ? lt.bottom <= clip.bottom + 1 : null,
						lastTurnTop: lt ? Math.round(lt.top - clip.top) : null,
					}
				: null,
			transcript: flat(document.querySelector("[data-lo-transcript-content]")),
			transcriptBusy:
				document.querySelector("[data-lo-canonical-transcript] [data-lo-streaming]") !== null ||
				document.querySelector("[data-lo-canonical-transcript] [data-lo-working-line]") !== null,
			stop: document.querySelector('button[aria-label="Stop agent"]') !== null,
		};
	})()`);
}

/** Where on screen a sentence landed (panel, composer band outside the panel, transcript, toast, elsewhere). */
function uxWhere(cdp, needle) {
	return cdp.evaluate(`(() => {
		const needle = ${JSON.stringify(needle)};
		const out = [];
		const walk = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
		let n = walk.nextNode();
		while (n) {
			if (n.nodeValue && n.nodeValue.includes(needle)) {
				const el = n.parentElement;
				const zone = el.closest("[data-lo-aside-panel]")
					? "panel"
					: el.closest("[data-lo-transcript-content]")
						? "transcript"
						: el.closest("[data-sonner-toast]")
							? "toast"
							: el.closest("[data-lo-composer-band]")
								? "composer-band"
								: "elsewhere";
				const roleEl = el.closest("[role]");
				const live = el.closest("[aria-live]");
				const r = el.getBoundingClientRect();
				out.push({
					zone,
					tag: el.tagName.toLowerCase(),
					role: roleEl ? roleEl.getAttribute("role") : null,
					live: live ? live.getAttribute("aria-live") : null,
					visible: r.width > 0 && r.height > 0,
					text: el.textContent.replace(/\\s+/g, " ").trim().slice(0, 240),
				});
			}
			n = walk.nextNode();
		}
		return out;
	})()`);
}

/** The Tab walk from the composer, `steps` presses in one direction. */
async function uxTabWalk(cdp, steps, shift) {
	const seen = [];
	for (let i = 0; i < steps; i++) {
		await pressChord(cdp, {
			key: "Tab",
			code: "Tab",
			virtualKeyCode: 9,
			modifiers: shift ? MODIFIER.shift : 0,
		});
		await wait(120);
		seen.push((await uxRead(cdp)).active);
	}
	return seen;
}
