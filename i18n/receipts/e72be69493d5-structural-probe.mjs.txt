import { execFileSync } from "node:child_process";
import { createHash } from "node:crypto";
import { createRequire } from "node:module";
import { resolve } from "node:path";
const require = createRequire(resolve("package.json"));
const { IntlMessageFormat } = require("intl-messageformat");
const { parse, TYPE } = require("@formatjs/icu-messageformat-parser");
const R5 = "5405030d0270b1aeddfe0366bed3f5200800b1cb";
const HEAD = "318ce5c4804c5d736b8c6f77b44a5735640a55db";
const L = ["es", "fr", "ru", "hi", "ur", "vi", "zh-CN"];
const show = (rev, p) => execFileSync("git", ["show", `${rev}:${p}`]);
const sha = (b) => createHash("sha256").update(b).digest("hex");
const BIDI = /[\u200E\u200F\u202A-\u202E\u2066-\u2069]/;
let fail = 0;
const enp = "src/i18n/catalogues/en/ui.mesh.json";
const enR5 = show(R5, enp), enH = show(HEAD, enp);
console.log(`en ui.mesh r5=${sha(enR5)} head=${sha(enH)} identical=${enR5.equals(enH)}`);
if (!enR5.equals(enH)) fail++;
const en = JSON.parse(enH);
const ctxp = "src/i18n/catalogues/en/ui.mesh.context.json";
const cB = JSON.parse(show(R5, ctxp)), cH = JSON.parse(show(HEAD, ctxp));
const ctxChanged = Object.keys({ ...cB, ...cH }).filter((k) => JSON.stringify(cB[k]) !== JSON.stringify(cH[k])).sort();
console.log(`context r5=${sha(show(R5, ctxp))} head=${sha(show(HEAD, ctxp))} changed=${JSON.stringify(ctxChanged)}`);
const ctxExpect = ["ui.mesh.approval.duration_hours", "ui.mesh.approval.duration_minutes", "ui.mesh.approval.duration_under_minute", "ui.mesh.approval.named_refusal"];
if (JSON.stringify(ctxChanged) !== JSON.stringify(ctxExpect)) fail++;
for (const k of ctxChanged) for (const f of ["role", "tone"]) if (cB[k][f] !== cH[k][f]) { console.log(`  context ${k}.${f} changed`); fail++; }
const cat = {};
console.log("\n(c) locale bytes, R5 -> HEAD");
for (const l of L) {
  const p = `src/i18n/catalogues/${l}/ui.mesh.json`;
  const b = show(R5, p), h = show(HEAD, p);
  const B = JSON.parse(b), H = JSON.parse(h);
  cat[l] = H;
  const changed = Object.keys({ ...B, ...H }).filter((k) => B[k] !== H[k]).sort();
  const keysEq = JSON.stringify(Object.keys(H).sort()) === JSON.stringify(Object.keys(en).sort());
  const expect = l === "ru" ? ["ui.mesh.approval.expires_window"] : [];
  const ok = keysEq && JSON.stringify(changed) === JSON.stringify(expect);
  // A line-level diff proves nothing else moved (formatting, ordering, trailing newline).
  const diffLines = execFileSync("git", ["diff", "--numstat", R5, HEAD, "--", p]).toString().trim() || "0\t0\t" + p;
  console.log(`${ok ? "PASS" : "FAIL"} ${l}: r5=${sha(b)} head=${sha(h)} bytes_identical=${b.equals(h)} keyset_matches_en=${keysEq} changed=${JSON.stringify(changed)} numstat=${diffLines.split("\t").slice(0, 2).join("/")}`);
  for (const k of changed) console.log(`    ${k}\n      r5:   ${B[k]}\n      head: ${H[k]}`);
  if (!ok) fail++;
}
// Structural checks on the one changed value.
const v = cat.ru["ui.mesh.approval.expires_window"];
const args = new Set(); for (const n of parse(v, { ignoreTag: true })) if (n.type === TYPE.argument) args.add(n.value);
console.log(`\nru expires_window structural: args=${[...args]} en_args=duration nfc=${v === v.normalize("NFC")} bidi_free=${!BIDI.test(v)}`);
if ([...args].join() !== "duration" || v !== v.normalize("NFC") || BIDI.test(v)) fail++;
// Mirror of approvalAnswerWindowLabel (mesh-approvals.ts) for one locale.
const fmt = (l, k, a) => new IntlMessageFormat(cat[l][k], l).format(a);
function windowLabel(l, seconds, outer) {
  const minutes = Math.ceil(seconds / 60);
  const duration = seconds < 60 ? fmt(l, "ui.mesh.approval.duration_under_minute")
    : minutes < 90 ? fmt(l, "ui.mesh.approval.duration_minutes", { count: minutes })
    : fmt(l, "ui.mesh.approval.duration_hours", { count: Math.ceil(minutes / 60) });
  return fmt(l, outer, { duration });
}
const pr = new Intl.PluralRules("ru");
console.log("\n(a) ru expires_window, every reachable duration (category: rendered)");

}

}
